# Reclaim Implementation Plan (G5 Wave 3, ability 1)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A corpse-gated radius revive Manifestation — revive fallen allies at 50% HP with their unlooted gear, holding 4 Resonance while they live and bleeding them out on the caster's death.

**Architecture:** Self/instant press-tick ability cast in `_rollback_tick` (like Lifebloom). Every cross-node effect uses the **plain-pending + own-tick apply** pattern (the G6 corpse-drag lesson — direct rollback-state writes across nodes are discarded). Held reservation reconciles like Lifebloom/Tangle; caster-death bleed-out is a victim-side pull like Tangle.

**Tech Stack:** Godot 4.7.1, GDScript (typed), GdUnit4, netfox rollback.

**Spec:** `docs/superpowers/specs/2026-07-20-reclaim-design.md` (read it — carries the full design).

## Global Constraints

- IP: original terms only (Resonance, Runners, Grid, Manifestations, Augments, Repulse, Glide) — never Shadowrun trademarks.
- All gameplay numbers via `MatchConfig.DEFAULTS`; baselines in `docs/ABILITY_TUNING_SHEET.md`.
- Typed GDScript; `class_name` for reusable classes.
- Tick-based simulation only: edge inputs are press-tick ints; new per-tick state registered in `RollbackSynchronizer.state_properties` (`fps_controller.tscn`); new inputs go through `PlayerInput` + `input_properties`.
- **Cross-node effects NEVER write another node's rollback state directly** — request via plain pending vars, apply in the target's own tick.
- Abilities are always-castable (no ownership gating), matching the prototype precedent.
- Null field suppresses casting (`_in_null_field()`); Wraith blocks casts.
- Tests tolerate failed Steam init (headless has no Steam). A test that spawns corpses must free the `corpses` group in `after_test` (G6 leak lesson).
- Commits end with `Co-Authored-By: Claude Opus 4.8 <noreply@anthropic.com>`.

```gdscript
func _cast_reclaim(tick: int, is_fresh: bool) -> void:
	var cost := MatchConfig.get_value("resonance.reclaim_cost")
	var targets := _reclaim_targets(MatchConfig.get_value("reclaim.radius_m"))
	if reclaim_reserved or _in_null_field() or resonance.current < cost or targets.is_empty():
		if is_fresh:
			cast_denied.emit()
		return
	for corpse in targets:
		var owner := _find_player_by_peer(corpse.owner_peer)
		if owner == null or not owner.is_dead:
			continue
		owner.queue_reclaim_revive(corpse.global_position,
			input.get_multiplayer_authority(), _corpse_snapshot(corpse))
		corpse.queue_reclaim_destroy()
	resonance.reserve(cost)
	reclaim_reserved = true
	reclaim_cast_tick = tick


func _reclaim_targets(radius: float) -> Array:
	var my_team := _reclaim_team_of(input.get_multiplayer_authority())
	var out: Array = []
	for node in get_tree().get_nodes_in_group("corpses"):
		var corpse := node as Corpse
		if corpse == null or corpse.removed:
			continue
		if corpse.global_position.distance_to(global_position) > radius:
			continue
		var owner := _find_player_by_peer(corpse.owner_peer)
		if owner == null or not owner.is_dead:
			continue
		if my_team != -1 and _reclaim_team_of(corpse.owner_peer) != my_team:
			continue
		out.append(corpse)
	return out


func _reclaim_team_of(peer: int) -> int:
	var ms := get_tree().get_first_node_in_group("match_state") as MatchState
	return ms.team_of(peer) if ms != null else -1


func _find_player_by_peer(peer: int) -> FPSController:
	for node in get_tree().get_nodes_in_group("players"):
		var p := node as FPSController
		if p != null and p.input.get_multiplayer_authority() == peer:
			return p
	return null


func _corpse_snapshot(corpse: Corpse) -> Array:
	return [corpse.w1_key, corpse.w1_mag, corpse.w1_reserve,
		corpse.w2_key, corpse.w2_mag, corpse.w2_reserve, corpse.grenades]


func queue_reclaim_revive(pos: Vector3, reviver_peer: int, snapshot: Array) -> void:
	_pending_reclaim_pos = pos
	_pending_reclaim_reviver = reviver_peer
	_pending_reclaim_snapshot = snapshot
	_has_pending_reclaim = true
```
